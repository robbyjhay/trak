A2. users
--------------------------------------------------------------------------------

model User {
  id                   String    @id @default(uuid()) @db.Uuid
  username             String    @unique @db.VarChar(64)
  email                String?   @unique @db.VarChar(255)
  passwordHash         String    @map("password_hash") @db.VarChar(255)
  role                 UserRole
  isSecretary          Boolean   @default(false) @map("is_secretary")
  isCorps              Boolean   @default(false) @map("is_corps")
  mustChangePassword   Boolean   @default(true) @map("must_change_password")
  emailVerifiedAt      DateTime? @map("email_verified_at") @db.Timestamptz(3)
  isActive             Boolean   @default(true) @map("is_active")
  lastLoginAt          DateTime? @map("last_login_at") @db.Timestamptz(3)
  createdAt            DateTime  @default(now()) @map("created_at") @db.Timestamptz(3)
  updatedAt            DateTime  @updatedAt @map("updated_at") @db.Timestamptz(3)

  profile              UserProfile?
  sessions             Session[]
  authTokens           AuthToken[]
  preferences          UserPreferences?

  @@map("users")
  @@index([role])
  @@index([isActive])
}

Constraints / validation:
  - username: required, 3–64 chars, unique, case-insensitive uniqueness
    enforced via citext OR store lowercase username_normalized
  - Phase 0 MUST add: usernameNormalized String @unique @map("username_normalized")
  - email: valid email format when present; unique when not null
  - passwordHash: bcrypt hash string; never null for active password users
  - role: enum head | member
  - isActive false → login rejected

A3. user_profiles
--------------------------------------------------------------------------------

model UserProfile {
  id            String   @id @default(uuid()) @db.Uuid
  userId        String   @unique @map("user_id") @db.Uuid
  name          String   @db.VarChar(200)
  phone         String   @default("") @db.VarChar(40)
  designation   String   @default("") @db.VarChar(200)
  gradeLevel    String   @default("") @map("grade_level") @db.VarChar(40)
  sex           String   @default("") @db.VarChar(32)
  stateOfOrigin String   @default("") @map("state_of_origin") @db.VarChar(100)
  dateJoined    DateTime? @map("date_joined") @db.Date
  corpsEnd      DateTime? @map("corps_end") @db.Date
  color         String   @default("#0e6b47") @db.VarChar(16)
  photoKey      String?  @map("photo_key") @db.VarChar(512)
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(3)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(3)

  user          User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("user_profiles")
  @@index([name])
}

Constraints:
  - 1:1 with users; ON DELETE CASCADE
  - name required non-empty
  - photoKey stores object storage key (not data URL)

A4. sessions
--------------------------------------------------------------------------------

model Session {
  id               String    @id @default(uuid()) @db.Uuid
  userId           String    @map("user_id") @db.Uuid
  tokenHash        String    @unique @map("token_hash") @db.VarChar(128)
  userAgent        String?   @map("user_agent") @db.VarChar(512)
  ipAddress        String?   @map("ip_address") @db.VarChar(64)
  expiresAt        DateTime  @map("expires_at") @db.Timestamptz(3)
  revokedAt        DateTime? @map("revoked_at") @db.Timestamptz(3)
  createdAt        DateTime  @default(now()) @map("created_at") @db.Timestamptz(3)
  lastUsedAt       DateTime  @default(now()) @map("last_used_at") @db.Timestamptz(3)

  user             User      @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("sessions")
  @@index([userId])
  @@index([expiresAt])
}

Constraints:
  - tokenHash = SHA-256 of opaque random token (32+ bytes)
  - Cookie stores raw token; DB stores hash only
  - Valid session: revokedAt IS NULL AND expiresAt > now()
  - ON DELETE CASCADE from user

A5. auth_tokens
--------------------------------------------------------------------------------

model AuthToken {
  id          String        @id @default(uuid()) @db.Uuid
  userId      String        @map("user_id") @db.Uuid
  type        AuthTokenType
  tokenHash   String        @unique @map("token_hash") @db.VarChar(128)
  expiresAt   DateTime      @map("expires_at") @db.Timestamptz(3)
  consumedAt  DateTime?     @map("consumed_at") @db.Timestamptz(3)
  createdAt   DateTime      @default(now()) @map("created_at") @db.Timestamptz(3)
  meta        Json?

  user        User          @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("auth_tokens")
  @@index([userId, type])
  @@index([expiresAt])
}

Constraints:
  - One-time use: set consumedAt on success
  - password_reset TTL 1 hour
  - invite TTL 7 days
  - email_verify TTL 48 hours
  - initial_password TTL 24 hours
